import { createClient } from "npm:@supabase/supabase-js@2.57.4";

/*
 * Bird WhatsApp delivery-status webhook receiver.
 *
 * Bird follows the Standard Webhooks specification:
 *   - Headers: webhook-id, webhook-timestamp, webhook-signature
 *   - Signature: HMAC-SHA256 over `{id}.{timestamp}.{rawBody}`, keyed with the
 *     base64-decoded endpoint secret (strip the `whsec_` prefix).
 *   - Dedup on webhook-id (at-least-once delivery).
 *
 * Event types we handle:
 *   whatsapp.accepted  → status = 'accepted'
 *   whatsapp.sent      → status = 'sent'
 *   whatsapp.delivered → status = 'delivered'
 *   whatsapp.read      → status = 'read'
 *   whatsapp.failed    → status = 'failed' + error fields → create delivery_alert + notify agent
 *   whatsapp.rejected  → status = 'rejected' + error fields → create delivery_alert + notify agent
 */

// --- Signature verification (Standard Webhooks) ---------------------------

async function verifySignature(
  rawBody: Uint8Array,
  headers: Headers,
): Promise<boolean> {
  if (!birdWebhookSecret) {
    console.error("BIRD_WEBHOOK_SECRET is not configured");
    return false;
  }

  const webhookId = headers.get("webhook-id");
  const webhookTimestamp = headers.get("webhook-timestamp");
  const webhookSignature = headers.get("webhook-signature");

  if (!webhookId || !webhookTimestamp || !webhookSignature) {
    return false;
  }

  // Reject timestamps older than 5 minutes
  const ageMs = Date.now() - parseInt(webhookTimestamp, 10) * 1000;
  if (isNaN(ageMs) || Math.abs(ageMs) > 5 * 60 * 1000) {
    console.error("Webhook timestamp out of range");
    return false;
  }

  // Strip whsec_ prefix and base64-decode to get key bytes
  let secret = birdWebhookSecret;
  if (secret.startsWith("whsec_")) {
    secret = secret.slice(6);
  }

  const keyBytes = base64ToBytes(secret);
  const message = new TextEncoder().encode(
    `${webhookId}.${webhookTimestamp}.`,
  );
  const combined = new Uint8Array(message.length + rawBody.length);
  combined.set(message, 0);
  combined.set(rawBody, message.length);

  const expectedSig = await hmacSha256Base64(keyBytes, combined);

  // During secret rotation, the header can contain multiple space-separated signatures
  // Each is prefixed with "v1,"
  return webhookSignature.split(" ").some((part) => {
    const sig = part.replace(/^v1,/, "");
    return timingSafeEqualBase64(sig, expectedSig);
  });
}

function base64ToBytes(b64: string): Uint8Array {
  const binary = atob(b64);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) {
    bytes[i] = binary.charCodeAt(i);
  }
  return bytes;
}

function bytesToBase64(bytes: Uint8Array): string {
  let binary = "";
  for (let i = 0; i < bytes.length; i++) {
    binary += String.fromCharCode(bytes[i]);
  }
  return btoa(binary);
}

async function hmacSha256Base64(key: Uint8Array, data: Uint8Array): Promise<string> {
  const cryptoKey = await crypto.subtle.importKey(
    "raw",
    key,
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"],
  );
  const sig = await crypto.subtle.sign("HMAC", cryptoKey, data);
  return bytesToBase64(new Uint8Array(sig));
}

function timingSafeEqualBase64(a: string, b: string): boolean {
  const aBytes = base64ToBytes(a);
  const bBytes = base64ToBytes(b);
  if (aBytes.length !== bBytes.length) return false;
  let diff = 0;
  for (let i = 0; i < aBytes.length; i++) {
    diff |= aBytes[i] ^ bBytes[i];
  }
  return diff === 0;
}

function formatTime(timeStr: string): string {
  const [h, m] = timeStr.split(":").map(Number);
  const period = h >= 12 ? "PM" : "AM";
  const hour12 = h === 0 ? 12 : h > 12 ? h - 12 : h;
  return `${hour12}:${m.toString().padStart(2, "0")} ${period}`;
}

function formatDate(dateStr: string): string {
  const [y, mo, d] = dateStr.split("-").map(Number);
  const date = new Date(y, mo - 1, d);
  return date.toLocaleDateString("en-US", {

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { status: 200, headers: corsHeaders });
  }

  if (req.method !== "POST") {
    return new Response(JSON.stringify({ error: "Method not allowed" }), {
      status: 405,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }

  try {
    // Capture raw body for signature verification
    const rawBodyArray = new Uint8Array(await req.arrayBuffer());

    // Verify signature
    const isValid = await verifySignature(rawBodyArray, req.headers);
    if (!isValid) {
      return new Response(
        JSON.stringify({ error: "Invalid signature" }),
        { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    // Parse body
    const bodyText = new TextDecoder().decode(rawBodyArray);
    const payload: BirdWebhookPayload = JSON.parse(bodyText);

    const webhookId = req.headers.get("webhook-id") || "";
    const eventType = payload.type || "";
    const birdMessageId = payload.data?.whatsapp_id || "";

    // Dedup on webhook-id
    const { data: existingEvent } = await supabase
      .from("webhook_events")
      .select("id")
      .eq("webhook_id", webhookId)
      .maybeSingle();

    if (existingEvent) {
      // Already processed — ack and return
      return new Response(JSON.stringify({ status: "duplicate", event_type: eventType }), {
        status: 200,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    // Record the webhook event
    await supabase.from("webhook_events").insert({
      webhook_id: webhookId,
      event_type: eventType,
      bird_message_id: birdMessageId,
    });

    // Find the attempt by Bird message ID
    if (!birdMessageId) {
      return new Response(JSON.stringify({ status: "no_message_id" }), {
        status: 200,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    const { data: attempt } = await supabase
      .from("reminder_attempts")
      .select("*")
      .eq("bird_message_id", birdMessageId)
      .maybeSingle();

    if (!attempt) {
      // No matching attempt — could be a message sent outside this system
      return new Response(JSON.stringify({ status: "no_attempt_found" }), {
        status: 200,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    const newStatus = STATUS_MAP[eventType];
    if (!newStatus) {
      // Unrecognized event type — ignore but ack
      return new Response(JSON.stringify({ status: "ignored", event_type: eventType }), {
        status: 200,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    // Build update for the attempt
    const update: Record<string, unknown> = {
      delivery_status: newStatus,
      status_timestamp: payload.timestamp || new Date().toISOString(),
    };

    // Capture error details for failure/rejected events
    if (FAILURE_TYPES.has(eventType) && payload.data?.error) {
      update.error_code = payload.data.error.code || null;
      update.error_description = payload.data.error.description || null;
      update.error_meta_code = payload.data.error.meta_error_code || null;
    }

    await supabase
      .from("reminder_attempts")
      .update(update)
      .eq("id", attempt.id);

    // For failed/rejected: create a delivery alert + notify agent (if not already notified)
    if (FAILURE_TYPES.has(eventType)) {
      // Check if alert already exists (dedup from duplicate webhooks handled above, but be safe)
      const { data: existingAlert } = await supabase
        .from("delivery_alerts")
        .select("id")
        .eq("attempt_id", attempt.id)
        .maybeSingle();

      if (!existingAlert) {
        await supabase.from("delivery_alerts").insert({
          attempt_id: attempt.id,
        });
      }

      // Notify agent if not already notified
      if (!attempt.agent_notified) {
        const notifResult = await notifyAgentOfFailure({
          id: attempt.id,
          appointment_id: attempt.appointment_id,
          recipient_address: attempt.recipient_address,
          error_code: (update.error_code as string) || attempt.error_code,
          error_description: (update.error_description as string) || attempt.error_description,
        });

        await supabase
          .from("reminder_attempts")
          .update({
            agent_notified: true,
            agent_notification_status: notifResult.status,
          })
          .eq("id", attempt.id);
      }
    }

    // For delivered/read: if this is a retry attempt, mark the parent as resolved
    if (newStatus === "delivered" && attempt.is_retry && attempt.parent_attempt_id) {
      await supabase
        .from("reminder_attempts")
        .update({ resolved_by_retry: true })
        .eq("id", attempt.parent_attempt_id);

      // Mark any alerts on the parent attempt as resolved
      await supabase
        .from("delivery_alerts")
        .update({ resolved: true, resolved_at: new Date().toISOString() })
        .eq("attempt_id", attempt.parent_attempt_id)
        .eq("resolved", false);
    }

    return new Response(JSON.stringify({ status: "processed", event_type: eventType, delivery_status: newStatus }), {
      status: 200,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  } catch (err) {
    console.error("Webhook handler error:", err);
    return new Response(
      JSON.stringify({ error: err instanceof Error ? err.message : "Unknown error" }),
      { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  }
});

function isDummyPhoneNumber(phone: string): boolean {
  const digits = phone.replace(/\D/g, "");
  if (digits.length < 10) return true;
  const last10 = digits.slice(-10);
  if (/^(\d)\1{9}$/.test(last10)) return true;
  if (last10 === "1234567890") return true;
  if (last10 === "0123456789") return true;
  if (/^55501\d{2}$/.test(last10)) return true;
  if (/^000\d{7}$/.test(last10)) return true;
  return false;
}

async function sendBirdSMS(toPhone: string, templateData: any, birdApiKey: string) {
  try {
    if (!birdFromNumber) {
      return { success: false, error: "BIRD_FROM_NUMBER is not configured in edge function secrets." };
    }
    const e164Pattern = /^\+?[1-9]\d{1,14}$/;
    const normalizedFrom = birdFromNumber.startsWith("+") ? birdFromNumber : `+${birdFromNumber}`;
    if (!e164Pattern.test(normalizedFrom)) {
      return { success: false, error: `BIRD_FROM_NUMBER is not in E.164 format: ${birdFromNumber}` };
    }

    const cleanNumber = toPhone.replace(/\D/g, "");
    const formattedTo = cleanNumber.startsWith("1") ? `+${cleanNumber}` : `+1${cleanNumber}`;

    const res = await fetch("https://us1.platform.bird.com/v1/whatsapp/messages", {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${birdApiKey}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        from: normalizedFrom,
        to: formattedTo,
        template: {
          slug: "appointment_confirmation",
          language: "en",
          components: [
            {
              type: "body",
              parameters: [
                { type: "text", name: "company_name", text: templateData.companyName },
                { type: "text", name: "first_name", text: templateData.firstName },
                { type: "text", name: "appointment_date", text: templateData.appointmentDate },
                { type: "text", name: "appointment_time", text: templateData.appointmentTime },
                { type: "text", name: "company_address", text: templateData.companyAddress },
                { type: "text", name: "callback_number", text: templateData.callbackNumber }
              ]
            }
          ]
        }
      }),
    });

    if (!res.ok) {
      const errText = await res.text();
      return { success: false, error: `Bird API error (${res.status}): ${errText}` };
    }
    // Bird returns 202 with a JSON body containing the message ID
    let birdMessageId: string | null = null;
    try {
      const resJson = await res.json();
      birdMessageId = resJson?.id || null;
    } catch {
      // Response was not JSON — proceed without message ID
    }
    return { success: true, birdMessageId };
  } catch (err: any) {
    return { success: false, error: err.message };
  }
}

async function sendEmail(
  to: string,
  subject: string,
  body: string,
  html: string,
): Promise<{ success: boolean; error?: string }> {
  if (!resendApiKey) {
    return { success: false, error: "RESEND_API_KEY is not configured in edge function secrets." };
  }

  try {
    const res = await fetch("https://api.resend.com/emails", {
      method: "POST",
      headers: {
        Authorization: `Bearer ${resendApiKey}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        from: "notifications@stressfreetaxes.com",
        to,
        subject,
        text: body,
        html: html,
      }),
    });

    if (!res.ok) {
      let errDetail = "";
      try {
        const errJson = await res.json();
        errDetail = errJson?.message || errJson?.error || JSON.stringify(errJson);
      } catch {
        try {
          errDetail = await res.text();
        } catch {
          errDetail = `HTTP ${res.status}`;
        }
      }
      return { success: false, error: `Resend API error (${res.status}): ${errDetail}` };
    }

    return { success: true };
  } catch (err) {
    return {
      success: false,
      error: err instanceof Error ? err.message : "Network error contacting Resend API.",
    };
  }
}

          { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } },
        );
      }

      // SMS channel via Bird API
      if (body.channel === "sms") {
        // Guard: skip if the customer has WhatsApp disabled or the recipient phone is a dummy/placeholder number.
        if (appt.send_whatsapp === false || isDummyPhoneNumber(body.recipient_address)) {

            JSON.stringify({ status: "skipped", channel: "sms", softFail: true }),
            { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } },
          );

          return new Response(
            JSON.stringify({ status: "failed", channel: "sms", error: "BIRD_API_KEY is not configured in edge function secrets." }),
            { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } },
          );

              channel: "sms",

            }),
            { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } },
          );

      .order("scheduled_send_at", { ascending: true })
      .limit(50);

      return new Response(JSON.stringify({ status: "failed", error: fetchError.message }), {
        status: 200,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    for (const reminder of (reminders || []) as ReminderRow[]) {

      if (!reminder.recipient_address) {

          .from("reminders")
          .update({ status: "skipped", error_details: "No recipient address", attempted_at: now })

        results.push({ id: reminder.id, status: "skipped" });
        continue;
      }

      // Fetch appointment details
      const { data: appt, error: apptError } = await supabase
        .from("appointments")
        .select("*")
        .eq("id", reminder.appointment_id)
        .maybeSingle<AppointmentRow>();

        await supabase
          .from("reminders")
          .update({ status: "failed", error_details: "Appointment not found", attempted_at: now, retry_count: reminder.retry_count + 1 })
          .eq("id", reminder.id);
        results.push({ id: reminder.id, status: "failed", error: "Appointment not found" });

        .from("profiles")
        .select("full_name, email, phone")
        .eq("id", appt.assigned_to)
        .maybeSingle<ProfileRow>();

          .from("reminders")
          .update({ status: "failed", error_details: "Coworker profile not found", attempted_at: now, retry_count: reminder.retry_count + 1 })
          .eq("id", reminder.id);
        results.push({ id: reminder.id, status: "failed", error: "Profile not found" });
        continue;
      }

      // Build email content
      const { subject, html, body } = reminder.recipient_type === "client"
        ? buildClientEmail(appt, profile)
        : buildCoworkerEmail(appt, profile);

      // Send email
      const result = await sendEmail(reminder.recipient_address, subject, body, html);

      if (result.success) {
        await supabase
          .from("reminders")
          .update({ status: "sent", attempted_at: now, error_details: "" })
          .eq("id", reminder.id);
        results.push({ id: reminder.id, status: "sent" });
      } else {
        // Retry logic: up to 3 retries
        const newRetryCount = reminder.retry_count + 1;
        if (newRetryCount >= 3) {
          await supabase
            .from("reminders")
            .update({ status: "failed", error_details: result.error || "Unknown error", attempted_at: now, retry_count: newRetryCount })
            .eq("id", reminder.id);
        } else {
          // Keep pending for retry, add backoff
          const backoffMinutes = Math.pow(2, newRetryCount) * 5;
          const nextAttempt = new Date(Date.now() + backoffMinutes * 60000).toISOString();
          await supabase
            .from("reminders")
            .update({ error_details: result.error || "Unknown error", attempted_at: now, retry_count: newRetryCount, scheduled_send_at: nextAttempt })
            .eq("id", reminder.id);
        }
        results.push({ id: reminder.id, status: "failed", error: result.error });
      }
    }

    return new Response(JSON.stringify({ processed: results.length, results }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  } catch (err) {
    return new Response(
      JSON.stringify({
        status: "failed",
        error: err instanceof Error ? err.message : "Unknown runtime error in edge function.",
      }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  }

      return new Response(JSON.stringify({ error: error.message }), {
        status: 400,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    const publicUrl = `${Deno.env.get("SUPABASE_URL")}/storage/v1/object/public/email-assets/${filename}`;

    return new Response(JSON.stringify({ success: true, publicUrl }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  } catch (err) {
    return new Response(
      JSON.stringify({ error: err instanceof Error ? err.message : "Unknown error" }),
      { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  }
});
