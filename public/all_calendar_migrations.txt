/*
# Shared Calendar Application — Initial Schema

## Overview
Creates the full database schema for a shared calendar and appointment management application
for coworkers who schedule Tax/Accounting, ACA, Medicare, Life Insurance, and Other appointments.

## New Tables
1. `profiles` — Extends auth.users with user data (email, name, phone, color, role, status)
2. `companies` — Company profiles with address and phone for client email reminders
3. `appointment_type_mappings` — Maps each appointment type to a company
4. `appointments` — Calendar appointments with client info, scheduling, and company association
5. `app_settings` — Singleton configuration (business timezone)

## Security
- RLS enabled on all tables
- Exclusion constraint prevents same-coworker appointment overlaps (concurrency-safe)
- end_time is always calculated server-side from start_time + duration_minutes
- Inactive user login blocked via trigger on auth.sessions

## Important Notes
- Email stored on profiles table
- 12 predefined soft colors, unique among active users
- US phone format validation
- Time slots 07:00–18:30 start, end_time <= 19:00
- Durations: 30, 60, 90, 120 minutes
- Company snapshot (JSONB) frozen on each appointment
*/

CREATE OR REPLACE FUNCTION validate_us_phone(phone text)
RETURNS boolean AS $$
BEGIN
  IF phone IS NULL THEN
    RETURN true;
  END IF;
  RETURN phone ~ '^\(\d{3}\)\s?\d{3}-\d{4}$';
END;
$$ LANGUAGE plpgsql IMMUTABLE;

ALTER TABLE profiles ADD CONSTRAINT profiles_phone_check
  CHECK (validate_us_phone(phone));

RETURNS trigger AS $$

  VALUES (NEW.id, NEW.email, COALESCE(NEW.raw_user_meta_data->>'full_name', ''));
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================================
-- APPOINTMENT TRIGGER: Calculate end_time, time_range, company_snapshot
-- ============================================================================
CREATE OR REPLACE FUNCTION calculate_appointment_fields()
RETURNS trigger AS $$
DECLARE
  mapped_company_id uuid;
  company_record companies%ROWTYPE;
BEGIN
  -- Calculate end_time from start_time + duration
  NEW.end_time = (NEW.start_time + (NEW.duration_minutes || ' minutes')::interval)::time;

  -- Validate end_time <= 19:00
  IF NEW.end_time > '19:00:00' THEN
    RAISE EXCEPTION 'Appointment end time % exceeds the 7:00 PM boundary.', NEW.end_time;
  END IF;

  -- Calculate time_range for exclusion constraint
  NEW.time_range = tstzrange(
    (NEW.appointment_date + NEW.start_time)::timestamptz,
    (NEW.appointment_date + (NEW.start_time + (NEW.duration_minutes || ' minutes')::interval))::timestamptz,
    '[)'
  );

  -- If company_id not explicitly set, get from type mapping
  IF NEW.company_id IS NULL THEN
    SELECT company_id INTO mapped_company_id
    FROM appointment_type_mappings
    WHERE appointment_type = NEW.appointment_type;
    NEW.company_id = mapped_company_id;
  END IF;

  -- Snapshot company info
  IF NEW.company_id IS NOT NULL THEN
    SELECT * INTO company_record FROM companies WHERE id = NEW.company_id;
    IF FOUND THEN
      NEW.company_snapshot = jsonb_build_object(
        'name', company_record.name,
        'address_line1', company_record.address_line1,
        'address_line2', company_record.address_line2,
        'city', company_record.city,
        'state', company_record.state,
        'zip_code', company_record.zip_code,
        'phone', company_record.phone
      );
    END IF;
  END IF;

  -- Set last_modified_by
  IF NEW.last_modified_by IS NULL THEN
    NEW.last_modified_by = auth.uid();
  END IF;

  -- Set created_by on insert
  IF TG_OP = 'INSERT' AND NEW.created_by IS NULL THEN
    NEW.created_by = auth.uid();
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS appointments_calculate_before_insert ON appointments;

# Reminders and Audit Logs

## Overview
Adds the reminders table for tracking email reminders and the audit_logs table
for recording all important changes.

## New Tables
1. `reminders` — Email reminder records with scheduling, status, and retry tracking
2. `audit_logs` — Server-written audit trail of all important actions

## Security
- reminders: admin-only SELECT; INSERT/UPDATE via edge functions (service role bypasses RLS)
- audit_logs: admin-only SELECT; no client INSERT/UPDATE/DELETE (written only by triggers/edge functions)

## Important Notes
- Unique constraint on (appointment_id, recipient_type) prevents duplicate reminders
- Reminders cancelled when appointment is cancelled or rescheduled (handled by edge function)
- Audit logs written by SECURITY DEFINER triggers on appointment changes
*/

-- ============================================================================
-- AUDIT TRIGGER: Log appointment changes
-- ============================================================================
CREATE OR REPLACE FUNCTION log_appointment_changes()
RETURNS trigger AS $$
DECLARE
  actor_id uuid := auth.uid();
  action_str text;
  summary text;
  prev_json jsonb;
  new_json jsonb;
BEGIN
  IF TG_OP = 'INSERT' THEN
    action_str := 'appointment.create';
    summary := format('Created appointment for %s (%s) on %s', NEW.client_name, NEW.appointment_type, NEW.appointment_date);
    new_json := jsonb_build_object(
      'client_name', NEW.client_name,
      'appointment_type', NEW.appointment_type,
      'appointment_date', NEW.appointment_date,
      'start_time', NEW.start_time,
      'duration_minutes', NEW.duration_minutes,
      'status', NEW.status,
      'assigned_to', NEW.assigned_to
    );
    prev_json := NULL;
  ELSIF TG_OP = 'UPDATE' THEN
    action_str := 'appointment.update';
    summary := format('Updated appointment for %s', NEW.client_name);

    -- Detect specific changes for summary
    IF OLD.status != NEW.status THEN
      summary := format('Changed status from %s to %s for %s on %s', OLD.status, NEW.status, NEW.client_name, NEW.appointment_date);
    ELSIF OLD.assigned_to != NEW.assigned_to THEN
      summary := format('Reassigned appointment for %s', NEW.client_name);
    ELSIF OLD.appointment_date != NEW.appointment_date OR OLD.start_time != NEW.start_time THEN
      summary := format('Rescheduled appointment for %s to %s at %s', NEW.client_name, NEW.appointment_date, NEW.start_time);
    END IF;

    prev_json := jsonb_build_object(
      'client_name', OLD.client_name,
      'appointment_type', OLD.appointment_type,
      'appointment_date', OLD.appointment_date,
      'start_time', OLD.start_time,
      'duration_minutes', OLD.duration_minutes,
      'status', OLD.status,
      'assigned_to', OLD.assigned_to
    );
    new_json := jsonb_build_object(
      'client_name', NEW.client_name,
      'appointment_type', NEW.appointment_type,
      'appointment_date', NEW.appointment_date,
      'start_time', NEW.start_time,
      'duration_minutes', NEW.duration_minutes,
      'status', NEW.status,
      'assigned_to', NEW.assigned_to
    );
  ELSIF TG_OP = 'DELETE' THEN
    action_str := 'appointment.delete';
    summary := format('Deleted appointment for %s on %s', OLD.client_name, OLD.appointment_date);
    prev_json := jsonb_build_object(
      'client_name', OLD.client_name,
      'appointment_type', OLD.appointment_type,
      'appointment_date', OLD.appointment_date,
      'start_time', OLD.start_time,
      'status', OLD.status
    );
    new_json := NULL;
    INSERT INTO audit_logs (user_id, action_type, table_name, record_id, change_summary, previous_values, new_values)
    VALUES (actor_id, action_str, 'appointments', OLD.id, summary, prev_json, new_json);
    RETURN OLD;
  END IF;

  INSERT INTO audit_logs (user_id, action_type, table_name, record_id, change_summary, previous_values, new_values)
  VALUES (actor_id, action_str, 'appointments', NEW.id, summary, prev_json, new_json);

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================================
-- REMINDER MANAGEMENT FUNCTION
-- Called by edge function to create/cancel reminders when appointments change
-- ============================================================================
CREATE OR REPLACE FUNCTION upsert_reminders(appt_id uuid)
RETURNS void AS $$
DECLARE
  appt appointments%ROWTYPE;
  coworker_profile profiles%ROWTYPE;
  send_time timestamptz;
BEGIN
  SELECT * INTO appt FROM appointments WHERE id = appt_id;
  IF NOT FOUND THEN
    -- Appointment deleted, cancel all pending reminders
    UPDATE reminders SET status = 'cancelled', error_details = 'Appointment deleted'
    WHERE appointment_id = appt_id AND status = 'pending';
    RETURN;
  END IF;

  -- If cancelled, cancel pending reminders
  IF appt.status = 'Cancelled' THEN
    UPDATE reminders SET status = 'cancelled', error_details = 'Appointment cancelled'
    WHERE appointment_id = appt_id AND status = 'pending';
    RETURN;
  END IF;

  -- If not scheduled, cancel pending reminders
  IF appt.status != 'Scheduled' THEN
    UPDATE reminders SET status = 'cancelled', error_details = 'Appointment not in Scheduled state'
    WHERE appointment_id = appt_id AND status = 'pending';
    RETURN;
  END IF;

  -- Calculate 8:00 AM ET on appointment date
  -- America/New_York is the configured timezone
  send_time = (appt.appointment_date::text || ' 08:00:00')::timestamp AT TIME ZONE 'America/New_York';

  -- Get coworker profile
  SELECT * INTO coworker_profile FROM profiles WHERE id = appt.assigned_to;
  IF NOT FOUND THEN RETURN; END IF;

  -- Upsert coworker reminder
  IF coworker_profile.email IS NOT NULL AND coworker_profile.email != '' THEN
    INSERT INTO reminders (appointment_id, recipient_type, recipient_address, scheduled_send_at, status)
    VALUES (appt_id, 'coworker', coworker_profile.email, send_time, 'pending')
    ON CONFLICT (appointment_id, recipient_type) DO UPDATE
    SET scheduled_send_at = EXCLUDED.scheduled_send_at,
        status = CASE WHEN reminders.status IN ('sent', 'failed') THEN reminders.status ELSE 'pending' END,
        error_details = CASE WHEN reminders.status IN ('sent', 'failed') THEN reminders.error_details ELSE '' END;
  END IF;

  -- Upsert client reminder
  IF appt.client_email IS NOT NULL AND appt.client_email != '' THEN
    INSERT INTO reminders (appointment_id, recipient_type, recipient_address, scheduled_send_at, status)
    VALUES (appt_id, 'client', appt.client_email, send_time, 'pending')
    ON CONFLICT (appointment_id, recipient_type) DO UPDATE
    SET scheduled_send_at = EXCLUDED.scheduled_send_at,
        recipient_address = EXCLUDED.recipient_address,
        status = CASE WHEN reminders.status IN ('sent', 'failed') THEN reminders.status ELSE 'pending' END,
        error_details = CASE WHEN reminders.status IN ('sent', 'failed') THEN reminders.error_details ELSE '' END;
  ELSE
    -- Mark client reminder as skipped if no email
    INSERT INTO reminders (appointment_id, recipient_type, recipient_address, scheduled_send_at, status, error_details)
    VALUES (appt_id, 'client', '', send_time, 'skipped', 'No client email on appointment')
    ON CONFLICT (appointment_id, recipient_type) DO UPDATE
    SET status = CASE WHEN reminders.status IN ('sent', 'failed') THEN reminders.status ELSE 'skipped' END,
        error_details = CASE WHEN reminders.status IN ('sent', 'failed') THEN reminders.error_details ELSE 'No client email on appointment' END;
  END IF;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================================
-- TRIGGER: Auto-manage reminders after appointment changes
-- ============================================================================
CREATE OR REPLACE FUNCTION manage_reminders_after_change()
RETURNS trigger AS $$
BEGIN
  PERFORM upsert_reminders(NEW.id);
  RETURN NEW;
END;

# Fix task-save failure: SECURITY DEFINER trigger cannot see auth.uid()

## Root cause
`record_task_status_change()` is SECURITY DEFINER (owned by postgres). Inside a
SECURITY DEFINER function, `auth.uid()` returns NULL because the request JWT is
not consulted in that execution context. The BEFORE INSERT trigger on `tasks`
called this function, which tried:

    INSERT INTO task_status_history (..., changed_by, ...)
    VALUES (..., auth.uid(), ...);

`changed_by` is `uuid NOT NULL`, so the NULL violated the constraint and
aborted the parent `tasks` INSERT — surfacing in the app as
"Failed to save task."

## Fix
1. In the INSERT branch, use `NEW.user_id` as the actor. The task row carries
   the owner (`user_id` defaults to `auth.uid()` from the session, set by the
   row-level default BEFORE the trigger reads NEW). This is the same value
   `auth.uid()` would have returned had it been available.
2. In the UPDATE branch, keep using `auth.uid()`. UPDATE triggers on a plain
   (non-SECURITY-DEFINER-needed) path — but to be safe and consistent, use
   `NEW.user_id` for the actor as well. The owner cannot change (UPDATE policy
   enforces user_id = auth.uid()), so `NEW.user_id` is the acting user.
3. Revoke/GRANT unchanged; function stays SECURITY DEFINER so it can write to
   task_status_history, which the authenticated role only has INSERT on via
   policy.

## Safety
- No data is deleted or renamed.
- No column types change.
- Function is replaced in place; triggers re-attached to existing definitions.
*/

CREATE OR REPLACE FUNCTION record_task_status_change()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  -- On INSERT: log the creation with from_status NULL.
  -- Use NEW.user_id as the actor: auth.uid() is NULL inside SECURITY DEFINER,
  -- but the tasks.user_id column defaults to auth.uid() at INSERT time, so
  -- NEW.user_id is the signed-in user who created the row.
  IF TG_OP = 'INSERT' THEN
    INSERT INTO task_status_history (task_id, from_status, to_status, changed_by, changed_at)
    VALUES (NEW.id, NULL, COALESCE(NEW.status, 'Not Started'), NEW.user_id, now());
    RETURN NEW;
  END IF;

  -- On UPDATE: only log when the status actually changed.
  -- NEW.user_id is the row owner; the UPDATE RLS policy guarantees the caller
  -- is that owner (auth.uid() = user_id), so it is the acting user.
  IF TG_OP = 'UPDATE' AND OLD.status IS DISTINCT FROM NEW.status THEN

- `assigned_user_id` (uuid, nullable) — the employee primarily responsible for this customer. REFERENCES `profiles(id) ON DELETE SET NULL` so deleting an employee un-links (but does not delete) the customer.
- `is_global` (boolean, NOT NULL, default false) — when true the customer is shared across all companies (not tied to a single company_id for filtering purposes).
- `send_whatsapp` (boolean, NOT NULL, default true) — controls whether WhatsApp/SMS reminders are sent to this customer.

2. Modified Tables
- `customers` — four new columns added via ADD COLUMN IF NOT EXISTS (idempotent). Existing rows get `is_global = false` and `send_whatsapp = true` defaults.

3. Security

4. Important Notes
